                try {
                    return JSON.parse(localStorage.getItem('ims_compare')) || [];
                } catch { return []; }
            });
            const [bookmarkedIds, setBookmarkedIds] = useState(() => {
                try {
                    return JSON.parse(localStorage.getItem('ims_bookmark')) || [];
                } catch { return []; }
            });

            // Sync localStorage
            useEffect(() => {
                localStorage.setItem('ims_compare', JSON.stringify(comparedIds));
            }, [comparedIds]);

            useEffect(() => {
                localStorage.setItem('ims_bookmark', JSON.stringify(bookmarkedIds));
            }, [bookmarkedIds]);

            // Toast helper
            const addToast = (message, isWarning = false) => {
                const id = Date.now();
                setToasts(prev => [...prev, { id, message, isWarning }]);
                setTimeout(() => {
                    setToasts(prev => prev.filter(t => t.id !== id));
                }, 3000);

                const totalScholarships = UNIVERSITY_DATABASE.filter(p => p.hasScholarship).length;
                return { totalPrograms, totalUnis, totalScholarships };
            }, []);

            const filteredPrograms = useMemo(() => {
                return UNIVERSITY_DATABASE.filter(item => {
                    // Saved filter mode
                    if (showSavedOnly && !bookmarkedIds.includes(item.id)) {
                        return false;
                    }

                    // Text Query
                    if (query.trim()) {
                        const q = query.toLowerCase();
                        const matchUni = item.university.toLowerCase().includes(q);
                        const matchFaculty = item.faculty.toLowerCase().includes(q);
                        const matchMajor = item.major.toLowerCase().includes(q);
                        const matchProgram = item.program.toLowerCase().includes(q);
                        const matchReq = item.requirements.toLowerCase().includes(q);
                        const matchScholar = item.scholarship.toLowerCase().includes(q);

                        if (!matchUni && !matchFaculty && !matchMajor && !matchProgram && !matchReq && !matchScholar) {
                            return false;
                        }
                    }

                    if (country && item.country !== country) return false;
                    if (faculty && item.faculty !== faculty) return false;
                    if (track && !item.track.includes(track)) return false;
                    if (intake && !item.intake.includes(intake)) return false;
                    if (scholarshipOnly && !item.hasScholarship) return false;

                    return true;
                }).sort((a, b) => {
                    if (sortBy === 'deadlineSoon') return new Date(a.deadline) - new Date(b.deadline);
                    if (sortBy === 'uniAsc') return a.university.localeCompare(b.university);
                    if (sortBy === 'majorAsc') return a.major.localeCompare(b.major);
                    return 0;
                });
            }, [query, country, faculty, track, intake, scholarshipOnly, sortBy, showSavedOnly, bookmarkedIds]);

            // Handlers

                setQuery("");
                setCountry("");
                setFaculty("");
                setTrack("");
                setIntake("");
                setScholarshipOnly(false);
                setShowSavedOnly(false);
                addToast("All search filters reset.");
            };

            const toggleCompare = (id) => {
                if (comparedIds.includes(id)) {
                    setComparedIds(prev => prev.filter(i => i !== id));
                    addToast("Removed program from comparison list.");
                } else {
                    if (comparedIds.length >= 4) {
                        addToast("Maximum 4 programs can be compared at once.", true);
                        return;
                    }
                    setComparedIds(prev => [...prev, id]);
                    addToast("Added program to comparison list.");
                }
            };

            const toggleBookmark = (id, e) => {
                if (e) e.stopPropagation();
                if (bookmarkedIds.includes(id)) {
                    setBookmarkedIds(prev => prev.filter(i => i !== id));
                    addToast("Removed from saved programs.");
                } else {
                    setBookmarkedIds(prev => [...prev, id]);
                    addToast("Saved program to bookmarks!");
                }
            };

                const val = parseFloat(studentGpa);
                if (isNaN(val) || val <= 0) {
                    setEligibilityResult({ isError: true, message: "Please enter a valid numeric score or GPA." });
                    return;
                }
                const target = activeDetailProgram ? activeDetailProgram.gpaRequirement : 3.0;
                if (val >= target || (val > 75 && target <= 4.0)) {
                    setEligibilityResult({ isMatch: true, message: `High Match Potential! Your score (${val}) meets or exceeds benchmark (~${target}).` });
                } else {
                    setEligibilityResult({ isMatch: false, message: `Moderate Match. Minimum estimated benchmark is ~${target}. Consider strengthening test scores.` });
                }
            };

            const handleExecuteHeroSearch = (searchTerm) => {
                setQuery(searchTerm);
                setShowSavedOnly(false);
                const el = document.getElementById('feedSection');
                if (el) el.scrollIntoView({ behavior: 'smooth' });
            };

                    {/* Header */}
                    <header className="sticky top-0 z-30 bg-white border-b border-slate-200 shadow-sm">
                        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
                            <div className="flex items-center justify-between h-20">
                                <div className="flex items-center space-x-4">
                                    <button onClick={clearAllFilters} className="text-left group focus:outline-none">
                                        <div className="flex items-center space-x-2">
                                            <span className="font-bold text-lg sm:text-xl text-slate-900 tracking-tight group-hover:text-brand-600 transition-colors">University Database 2026</span>

                                        <p className="text-xs text-slate-500 font-medium">SMA Khalifa IMS • Life Readiness Project (LRP) University Exploration Program (2025/2026)</p>

                                </div>

                                <div className="flex items-center space-x-3">
                                    <button 
                                        onClick={() => {
                                            if (comparedIds.length === 0) {
                                                addToast("Select at least 1 program to compare.", true);
                                                return;
                                            }
                                            setIsCompareModalOpen(true);
                                        }} 
                                        className="relative inline-flex items-center px-3.5 py-2 text-sm font-medium rounded-lg text-slate-700 bg-slate-100 hover:bg-slate-200 transition-colors"
                                    >
                                        <i className="fa-solid fa-code-compare mr-2 text-brand-600"></i>
                                        <span className="hidden md:inline">Compare</span>

                                    </button>

                                    <button 
                                        onClick={() => {
                                            if (!showSavedOnly && bookmarkedIds.length === 0) {
                                                addToast("No saved programs yet. Click the bookmark icon to save programs.", true);
                                                return;
                                            }
                                            setShowSavedOnly(!showSavedOnly);
                                        }} 
                                        className={`relative inline-flex items-center px-3.5 py-2 text-sm font-medium rounded-lg transition-colors ${showSavedOnly ? 'bg-amber-500 text-white' : 'text-slate-700 bg-slate-100 hover:bg-slate-200'}`}
                                    >
                                        <i className={`fa-solid fa-bookmark mr-2 ${showSavedOnly ? 'text-white' : 'text-amber-500'}`}></i>
                                        <span className="hidden md:inline">{showSavedOnly ? 'All Programs' : 'Saved'}</span>
                                        <span className={`ml-2 text-xs px-2 py-0.5 rounded-full font-bold ${showSavedOnly ? 'bg-white text-amber-600' : 'bg-amber-500 text-white'}`}>{bookmarkedIds.length}</span>
                                    </button>
                                </div>
                            </div>
                        </div>
                    </header>

                                    <button

                            <div className="grid grid-cols-2 md:grid-cols-4 gap-4 text-center divide-x divide-slate-100">
                                <div>
                                    <span className="block text-xl font-bold text-slate-900">{stats.totalPrograms}</span>
                                    <span className="text-xs text-slate-500 uppercase tracking-wider">Programs Indexed</span>
                                </div>
                                <div>
                                    <span className="block text-xl font-bold text-slate-900">{stats.totalUnis}</span>

                                </div>
                                <div>
                                    <span className="block text-xl font-bold text-brand-600">{stats.totalScholarships}</span>
